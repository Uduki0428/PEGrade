    <script type="text/babel">
        const { useState, useEffect, useMemo, useCallback } = React;

        // --- 1. 配置与标准数据 ---

        // 清空默认学生数据
        const INITIAL_STUDENTS = [];

            { name: "张老师", id: "T001" },

        // 评分标准表
        const SCORING_STANDARDS = {
            // 长跑使用 mm.ss 格式，需转换为秒
            '1200m跑': { type: 'long_run', gender: '女', table: [[5.14, 100], [5.40, 90], [6.07, 80], [6.33, 70], [7.00, 60], [7.15, 50], [7.30, 40], [8.30, 0]] },

            // 立定跳远 (米)
            '立定跳远_男': { type: 'high_better', gender: '男', table: [[2.73, 100], [2.57, 90], [2.41, 80], [2.26, 70], [2.10, 60], [2.05, 50], [2.00, 40], [1.80, 0]] },
            '立定跳远_女': { type: 'high_better', gender: '女', table: [[2.07, 100], [1.95, 90], [1.83, 80], [1.72, 70], [1.60, 60], [1.54, 50], [1.49, 40], [1.25, 0]] },
            // 短跑 (秒)
            '100m跑_男': { type: 'low_better', gender: '男', table: [[13.40, 100], [14.00, 90], [14.60, 80], [15.20, 70], [15.80, 60], [16.40, 50], [17.00, 40], [19.40, 0]] },
            '100m跑_女': { type: 'low_better', gender: '女', table: [[16.40, 100], [17.00, 90], [17.60, 80], [18.20, 70], [18.80, 60], [19.40, 50], [20.00, 40], [22.40, 0]] },
            // 力量 (个)
            '俯卧撑': { type: 'high_better', gender: '男', table: [[40, 100], [35, 90], [30, 80], [25, 70], [20, 60], [15, 50], [10, 40], [0, 0]] },
            '仰卧起坐': { type: 'high_better', gender: '女', table: [[56, 100], [49, 90], [42, 80], [35, 70], [28, 60], [26, 50], [24, 40], [14, 0]] }
        };

        // --- Helper Functions ---

        const parseInputToValue = (input, isLongRun) => {
            if (!input) return 0;
            const str = String(input);
            if (isLongRun) {
                const parts = str.split('.');
                const min = parseInt(parts[0] || 0);
                const sec = parseInt(parts[1] || 0);
                return min * 60 + sec;
            } else {
                return parseFloat(str);
            }
        };

        const calculateStandardScore = (inputVal, itemName, gender) => {
            let key = itemName;
            if (itemName === '立定跳远' || itemName === '100m跑') {
                key = `${itemName}_${gender}`;
            }
            
            const standard = SCORING_STANDARDS[key];
            if (!standard) return null;

            const isLongRun = standard.type === 'long_run';
            const val = parseInputToValue(inputVal, isLongRun);
            
            const table = standard.table.map(([threshold, score]) => {
                const tVal = isLongRun ? parseInputToValue(threshold, true) : threshold;
                return [tVal, score];
            });

            if (standard.type === 'high_better') {
                if (val >= table[0][0]) return 100; 
                if (val <= table[table.length - 1][0]) return 0; 
            } else {
                if (val <= table[0][0]) return 100;
                if (val >= table[table.length - 1][0]) return 0; 
            }

            for (let i = 0; i < table.length - 1; i++) {
                const [currentVal, currentScore] = table[i];
                const [nextVal, nextScore] = table[i+1];

                let inRange = false;
                if (standard.type === 'high_better') {
                    if (val <= currentVal && val >= nextVal) inRange = true;
                } else {
                    if (val >= currentVal && val <= nextVal) inRange = true;
                }

                if (inRange) {
                    const ratio = (val - currentVal) / (nextVal - currentVal);
                    const score = currentScore + ratio * (nextScore - currentScore);
                    return parseFloat(score.toFixed(1)); 
                }
            }
            return 0;
        };

        const getDefaultSemester = () => {
            const now = new Date();
            const month = now.getMonth(); 
            const year = now.getFullYear();
            if (month >= 8 || month <= 1) {
                const academicYear = month >= 8 ? year : year - 1;
                return `${academicYear}-${academicYear + 1}-1`;
            } else {
                return `${year - 1}-${year}-2`;
            }
        };

        const getSemesterOptions = () => {
            const now = new Date();
            const currentYear = now.getFullYear();
            const options = [];
            for (let i = -2; i <= 2; i++) {
                const y = currentYear + i;
                options.push(`${y}-${y + 1}-1`);
                options.push(`${y}-${y + 1}-2`);
            }
            return options;
        };

        const getSemesterType = (semStr) => {
            const lastChar = semStr.slice(-1);
            return lastChar === '1' ? 'fall' : 'spring';
        };

        const getTestItemsName = (semesterType, gender) => {
            if (semesterType === 'fall') {
                return { item1: gender === '男' ? '1500m跑' : '1200m跑', item2: '立定跳远' };
            } else {
                return { item1: gender === '男' ? '俯卧撑' : '仰卧起坐', item2: '100m跑' };
            }
        };

        const calculateFinalScore = (scores, semesterType) => {
            const { regular1 = 0, regular2 = 0, special1 = 0, special2 = 0, runScore = 0, stdItem1 = 0, stdItem2 = 0, theory = 0 } = scores;
            const specialAvg = (parseFloat(special1) + parseFloat(special2)) / 2;
            const stdTotal = (parseFloat(stdItem1) + parseFloat(stdItem2)) / 2;
            let total = 0;
            if (semesterType === 'fall') {
                total = (regular1 * 0.1) + (regular2 * 0.1) + (stdTotal * 0.3) + (runScore * 0.1) + (specialAvg * 0.4);
            } else {
                total = (regular1 * 0.05) + (regular2 * 0.05) + (theory * 0.1) + (stdTotal * 0.3) + (runScore * 0.1) + (specialAvg * 0.4);
            }
            return total.toFixed(1);
        };

            const [svgHtml, setSvgHtml] = useState('');
            useEffect(() => {
                if (window.lucide && window.lucide.icons) {
                    const pascalName = name.replace(/(^\w|-\w)/g, (c) => c.replace(/-/, "").toUpperCase());
                    const iconNode = window.lucide.icons[pascalName] || window.lucide.icons[name];
                    if (iconNode && iconNode.toSvg) {
                         setSvgHtml(iconNode.toSvg({ width: size, height: size, class: className }));
                    }
                }
            }, [name, size, className]);
            if (!svgHtml) return <span className={className}>{name}</span>;
            return <span style={{ display: 'contents' }} dangerouslySetInnerHTML={{ __html: svgHtml }} />;
        };

                    <span className="text-sm font-medium">{message}</span>

        // 统一登录页 - 修改顺序：姓名在前，工号在后
        const LoginPage = ({ onLogin, users }) => {
            const [nameInput, setNameInput] = useState('');
            const [idInput, setIdInput] = useState('');
            const [error, setError] = useState('');

            const handleLogin = (e) => {
                e.preventDefault();
                setError('');
                // 管理员判断逻辑：姓名输入“管理员”或“admin”，工号输入“admin888”

                            </div>
                            {error && <p className="text-red-500 text-sm text-center animate-pulse">{error}</p>}
                            <button type="submit" className="w-full bg-blue-600 hover:bg-blue-700 text-white font-bold py-2 px-4 rounded-lg transition-colors shadow-lg">登录</button>
                        </form>
                    </div>
                </div>
            );
        };

        // --- 抽离的打分弹窗组件 ---
        const GradingModal = ({ student, semester, onClose, onSave }) => {
            const [scores, setScores] = useState({ ...student.scores });
            const [rawInputs, setRawInputs] = useState({
                item1: student.scores.rawItem1 || '',
                item2: student.scores.rawItem2 || ''
            });

            const semType = getSemesterType(semester);
            const testNames = getTestItemsName(semType, student.gender);

            const updateScore = (field, value) => {
                setScores(prev => ({ ...prev, [field]: value }));
            };

            const handleRawInput = (itemKey, value, scoreField, itemName) => {
                setRawInputs(prev => ({ ...prev, [itemKey]: value }));
                const calcScore = calculateStandardScore(value, itemName, student.gender);
                if (calcScore !== null) {
                    updateScore(scoreField, calcScore);
                }
            };

            const handleSave = () => {
                onSave(student.id, { ...scores, rawItem1: rawInputs.item1, rawItem2: rawInputs.item2 });
                onClose();
            };

            return (
                <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-[100] p-4">
                    <div className="bg-white rounded-xl w-full max-w-lg shadow-2xl flex flex-col max-h-[90vh] relative z-[101]">

                                <div><label className="text-xs text-gray-500">平时1</label><input type="number" className="w-full border p-2 rounded" value={scores.regular1} onChange={(e) => updateScore('regular1', e.target.value)} /></div>
                                <div><label className="text-xs text-gray-500">平时2</label><input type="number" className="w-full border p-2 rounded" value={scores.regular2} onChange={(e) => updateScore('regular2', e.target.value)} /></div>
                            </div>

                                <div><label className="text-xs text-gray-500">专项1</label><input type="number" className="w-full border p-2 rounded" value={scores.special1} onChange={(e) => updateScore('special1', e.target.value)} /></div>
                                <div><label className="text-xs text-gray-500">专项2</label><input type="number" className="w-full border p-2 rounded" value={scores.special2} onChange={(e) => updateScore('special2', e.target.value)} /></div>

"如 5.40" : "个数"} className="w-full border border-blue-200 p-2 rounded focus:ring-2 ring-blue-400 outline-none" value={rawInputs.item1} onChange={(e) => handleRawInput('item1', e.target.value, 'stdItem1', testNames.item1)} />
                                        </div>
                                        <div>
                                            <label className="text-xs text-gray-600 block mb-1">折算分</label>

"如 8.6" : "米"} className="w-full border border-blue-200 p-2 rounded focus:ring-2 ring-blue-400 outline-none" value={rawInputs.item2} onChange={(e) => handleRawInput('item2', e.target.value, 'stdItem2', testNames.item2)} />
                                        </div>
                                        <div>
                                            <label className="text-xs text-gray-600 block mb-1">折算分</label>
                                            <input type="number" className="w-full border p-2 rounded bg-gray-100 text-blue-600 font-bold" value={scores.stdItem2} readOnly />
                                        </div>
                                    </div>
                                </div>
                            </div>

                            <div className="grid grid-cols-2 gap-4">
                                <div><label className="text-xs text-gray-500">乐跑次数</label><input type="number" className="w-full border p-2 rounded bg-gray-50" value={scores.runCount} onChange={(e) => updateScore('runCount', e.target.value)} /></div>
                                <div><label className="text-xs text-gray-500">乐跑得分</label><input type="number" className="w-full border p-2 rounded" value={scores.runScore} onChange={(e) => updateScore('runScore', e.target.value)} /></div>

                    </div>

        // --- 抽离的批量录入表格组件 ---
        const BatchEntryTable = ({ students, field, fieldName, semester, onBatchSave }) => {
            const semType = getSemesterType(semester);
            return (

                                    {/* 动态显示列头 */}
                                    {(field === 'stdItem1' || field === 'stdItem2') ? `${fieldName} (原始成绩 -> 折算分)` : `${fieldName} (输入后自动保存)`}
                                </th>
                            </tr>
                        </thead>
                        <tbody className="bg-white divide-y divide-gray-200">
                            {students.map(s => (
                                <BatchRow key={s.id} student={s} field={field} semester={semester} onSave={onBatchSave} />
                            ))}
                        </tbody>
                    </table>

            );

        // 单独的行组件 - 升级为支持统考自动换算
        const BatchRow = React.memo(({ student, field, semester, onSave }) => {
            // 判断是否是统考项目
            const isStdItem = field === 'stdItem1' || field === 'stdItem2';
            
            // 确定绑定的 Key: 统考绑定 rawItem1/2，普通绑定 field
            const rawKey = field === 'stdItem1' ? 'rawItem1' : (field === 'stdItem2' ? 'rawItem2' : null);
            const actualBindKey = isStdItem ? rawKey : field;
            
            const [val, setVal] = useState(student.scores[actualBindKey]);

            // 当切换列或数据源更新时，重置本地状态
            useEffect(() => {
                setVal(student.scores[actualBindKey]);
            }, [actualBindKey, student.scores]);

            const handleChange = (e) => {
                const newValue = e.target.value;
                setVal(newValue);
                
                if (isStdItem) {
                    // 如果是统考，需要计算分数并更新两个字段 (原始值 + 分数)
                    const semType = getSemesterType(semester);
                    const testNames = getTestItemsName(semType, student.gender);
                    const itemName = field === 'stdItem1' ? testNames.item1 : testNames.item2;
                    
                    const calcScore = calculateStandardScore(newValue, itemName, student.gender);
                    
                    // 保存：同时更新 原始值(rawKey) 和 分数(field)
                    onSave(student.id, { 
                        [rawKey]: newValue,
                        [field]: calcScore !== null ? calcScore : 0
                    });
                } else {
                    // 普通项目，只更新当前字段
                    onSave(student.id, { [field]: newValue });
                }
            };

            return (
                <tr>
                    <td className="px-6 py-4 whitespace-nowrap text-sm font-medium text-gray-900">{student.name}</td>
                    <td className="px-6 py-4 whitespace-nowrap text-sm text-gray-500">{student.id}</td>
                    <td className="px-6 py-4 whitespace-nowrap flex items-center gap-2">
                        <input 
                            type={isStdItem ? "text" : "number"}
                            placeholder={isStdItem ? "原始成绩" : ""}
                            className="input-cell w-32 border border-gray-300 rounded px-2 py-1"
                            value={val || ''}
                            onChange={handleChange}
                        />
                        {/* 如果是统考项目，显示折算分 */}
                        {isStdItem && (
                            <span className="text-sm font-bold text-blue-600 bg-blue-50 px-2 py-1 rounded">
                                {student.scores[field] || 0}分
                            </span>
                        )}
                    </td>
                </tr>
            );
        });

        // 管理员面板
        const AdminPanel = ({ students, setStudents, teachers, setTeachers, semester, setSemester, onLogout }) => {
            const [activeTab, setActiveTab] = useState('data');
            const semesterOptions = getSemesterOptions();

            const handleFileUpload = (e, type) => {
                const file = e.target.files?.[0];
                if (!file) return;
                const reader = new FileReader();
                reader.onload = (evt) => {
                    const bstr = evt.target.result;
                    const wb = XLSX.read(bstr, { type: 'binary' });
                    const wsname = wb.SheetNames[0];
                    const ws = wb.Sheets[wsname];
                    const data = XLSX.utils.sheet_to_json(ws);
                    
                    if (data.length === 0) {
                        alert("表格为空，请检查");
                        return;
                    }

                    const firstRow = data[0];
                    const requiredCols = type === 'students' ? ['学号', '姓名'] : type === 'run' ? ['学号'] : ['姓名', '工号'];
                    const missing = requiredCols.filter(col => !Object.keys(firstRow).includes(col));
                    
                    if (missing.length > 0) {
                        alert(`导入失败：表格缺少以下必要列：${missing.join(', ')}`);
                        e.target.value = ''; 
                        return;
                    }

                    if (type === 'students') processStudentData(data);
                    else if (type === 'run') processRunData(data);
                    else if (type === 'teachers') processTeacherData(data);
                    e.target.value = '';
                };
                reader.readAsBinaryString(file);
            };

            const processStudentData = (data) => {
                const newStudents = data.map(row => ({
                    id: String(row['学号'] || row['id']),
                    name: row['姓名'],
                    gender: row['性别'],
                    adminClass: row['行政班级'],
                    optionClass: row['体育课选项班'] || row['上课班级'] || '未分配班级',
                    teacherName: row['体育课任课教师'] || '',
                    teacherId: '', 
                    scores: { regular1:0, regular2:0, special1:0, special2:0, runCount:0, runScore:0, stdItem1:0, stdItem2:0, theory:0 }
                }));
                const merged = [...students];
                let addedCount = 0;
                newStudents.forEach(ns => {
                    const idx = merged.findIndex(s => s.id === ns.id);
                    if (idx >= 0) { 
                        merged[idx] = { ...merged[idx], ...ns, scores: merged[idx].scores }; 
                    } else { 
                        merged.push(ns); 
                        addedCount++;
                    }
                });
                setStudents(merged);
                alert(`处理完成：新增 ${addedCount} 人，更新 ${newStudents.length - addedCount} 人`);
            };

            const processRunData = (data) => {
                let count = 0;
                const updatedStudents = students.map(s => {
                    const record = data.find(r => String(r['学号']) === s.id);
                    if (record) {
                        count++;
                        return { ...s, scores: { ...s.scores, runCount: record['乐跑总次数'] || 0, runScore: record['达成度评价表录入百分制成绩'] || 0 } };
                    }
                    return s;
                });
                setStudents(updatedStudents);
                alert(`成功匹配并更新 ${count} 名学生的乐跑成绩`);
            };

            const processTeacherData = (data) => {
                const newTeachers = data.map(r => ({ name: r['姓名'], id: String(r['工号']) }));
                setTeachers(newTeachers);
                alert(`导入 ${newTeachers.length} 位教师名单`);
            };

            return (
                <div className="min-h-screen bg-gray-50">
                    <header className="bg-white shadow">
                        <div className="max-w-7xl mx-auto px-4 py-4 flex justify-between items-center">
                            <h1 className="text-xl font-bold text-gray-800">管理员控制台</h1>
                            <div className="flex items-center gap-4">
                                <span className="text-sm text-gray-500">设置学期:</span>
                                <select value={semester} onChange={(e) => setSemester(e.target.value)} className="border rounded px-2 py-1 text-sm bg-white">
                                    {semesterOptions.map(opt => (<option key={opt} value={opt}>{opt} ({opt.endsWith('1') ? '秋季' : '春季'})</option>))}
                                </select>
                                <button onClick={onLogout} className="text-red-600 text-sm hover:underline">退出</button>
                            </div>
                        </div>
                    </header>
                    <main className="max-w-7xl mx-auto px-4 py-6">
                        <div className="flex space-x-4 mb-6 overflow-x-auto">
                            {['data', 'teachers', 'run'].map(tab => (
                                <button key={tab} onClick={() => setActiveTab(tab)} className={`px-4 py-2 rounded-lg font-medium whitespace-nowrap ${activeTab === tab ? 'bg-blue-600 text-white' : 'bg-white text-gray-600 hover:bg-gray-100'}`}>
                                    {tab === 'data' && '学生基础数据'} {tab === 'teachers' && '教师管理'} {tab === 'run' && '乐跑成绩导入'}
                                </button>
                            ))}
                        </div>
                        <div className="bg-white rounded-xl shadow p-6">
                            {activeTab === 'data' && (
                                <div className="space-y-6">
                                    <div className="border-b pb-4">
                                        <h3 className="font-semibold mb-2">步骤1: 导入学生名单</h3>
                                        <p className="text-sm text-gray-500 mb-2">支持列名：[体育课选项班] 或 [上课班级]</p>
                                        <input type="file" accept=".xlsx, .xls" onChange={(e) => handleFileUpload(e, 'students')} className="block w-full text-sm text-gray-500 file:mr-4 file:py-2 file:px-4 file:rounded-full file:border-0 file:text-sm file:font-semibold file:bg-blue-50 file:text-blue-700 hover:file:bg-blue-100"/>
                                    </div>
                                    <div className="max-h-96 overflow-y-auto border rounded text-sm">
                                        <table className="min-w-full divide-y divide-gray-200">
                                            <thead className="bg-gray-50 sticky top-0"><tr><th className="px-4 py-2 text-left">学号</th><th className="px-4 py-2 text-left">姓名</th><th className="px-4 py-2 text-left">选项班</th><th className="px-4 py-2 text-left">任课教师</th></tr></thead>
                                            <tbody className="bg-white divide-y divide-gray-200">{students.slice(0, 100).map(s => (<tr key={s.id}><td className="px-4 py-2">{s.id}</td><td className="px-4 py-2">{s.name}</td><td className="px-4 py-2">{s.optionClass}</td><td className="px-4 py-2 text-gray-500">{s.teacherName || '未分配'}</td></tr>))}</tbody>
                                        </table>
                                    </div>
                                </div>
                            )}
                            {activeTab === 'teachers' && (
                                <div className="space-y-6">
                                    <div className="border-b pb-4"><h3 className="font-semibold mb-2">导入教师名单 (可选)</h3><input type="file" accept=".xlsx, .xls" onChange={(e) => handleFileUpload(e, 'teachers')} className="block w-full text-sm text-gray-500 file:mr-4 file:py-2 file:px-4 file:rounded-full file:border-0 file:text-sm file:font-semibold file:bg-blue-50 file:text-blue-700 hover:file:bg-blue-100"/></div>
                                    <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">{teachers.map(t => (<div key={t.id} className="border p-3 rounded flex justify-between items-center"><span>{t.name}</span><span className="bg-gray-100 px-2 py-1 rounded text-xs">{t.id}</span></div>))}</div>
                                </div>
                            )}
                            {activeTab === 'run' && (
                                <div className="space-y-6">
                                    <div className="bg-yellow-50 border-l-4 border-yellow-400 p-4"><p className="text-sm text-yellow-700">Excel需包含列：<strong>学号</strong>、<strong>乐跑总次数</strong>、<strong>达成度评价表录入百分制成绩</strong></p></div>
                                    <input type="file" accept=".xlsx, .xls" onChange={(e) => handleFileUpload(e, 'run')} className="block w-full text-sm text-gray-500 file:mr-4 file:py-2 file:px-4 file:rounded-full file:border-0 file:text-sm file:font-semibold file:bg-blue-50 file:text-blue-700 hover:file:bg-blue-100"/>
                                </div>
                            )}
                        </div>
                    </main>
                </div>
            );
        };

        // 教师面板
        const TeacherPanel = ({ currentUser, students, setStudents, semester, setSemester, onLogout }) => {
            const [selectedClass, setSelectedClass] = useState('');
            const [editingStudent, setEditingStudent] = useState(null);
            const [showClaimModal, setShowClaimModal] = useState(false);
            const [toastMsg, setToastMsg] = useState('');
            
            const [viewMode, setViewMode] = useState('card'); 
            const [batchField, setBatchField] = useState('regular1'); 
            
            const semType = getSemesterType(semester);
            const semesterOptions = getSemesterOptions();

            const myStudents = useMemo(() => students.filter(s => s.teacherName === currentUser.name), [students, currentUser.name]);
            
            const myClasses = useMemo(() => {
                const classes = [...new Set(myStudents.map(s => s.optionClass))];
                return classes.sort((a, b) => a.localeCompare(b, 'zh-CN'));
            }, [myStudents]);

            const classStudents = useMemo(() => {
                if (!selectedClass) return [];
                return myStudents.filter(s => s.optionClass === selectedClass).sort((a, b) => parseInt(a.id) - parseInt(b.id));
            }, [myStudents, selectedClass]);

            const allClassesForClaim = useMemo(() => {
                const map = {};
                students.forEach(s => {
                    const clsName = s.optionClass;
                    if (!map[clsName]) map[clsName] = { name: clsName, teacher: s.teacherName, count: 0 };
                    map[clsName].count++;
                    if(s.teacherName && !map[clsName].teacher) map[clsName].teacher = s.teacherName;
                });
                return Object.values(map).sort((a, b) => a.name.localeCompare(b.name, 'zh-CN'));
            }, [students]);

            const handleSaveFullScore = (studentId, newScores) => {
                setStudents(prev => prev.map(s => s.id === studentId ? { ...s, scores: newScores } : s));
            };

            // 修改: 支持同时更新多个字段 (updates 是对象)
            const handleBatchSaveScore = useCallback((studentId, updates) => {
                setStudents(prev => prev.map(s => {
                    if (s.id === studentId) {
                        return { ...s, scores: { ...s.scores, ...updates } };
                    }
                    return s;
                }));
            }, []); 

            const handleClaimClass = (className) => {
                const updatedStudents = students.map(s => s.optionClass === className ? { ...s, teacherName: currentUser.name, teacherId: currentUser.id } : s);
                setStudents(updatedStudents);
                setToastMsg(`认领成功: "${className}"`);
            };

            const handleExport = () => {
                if (!selectedClass) return;
                const exportData = classStudents.map((s, index) => {
                    const finalScore = calculateFinalScore(s.scores, semType);
                    const row = {
                        '序号': index + 1, '行政班级': s.adminClass, '学号': s.id, '姓名': s.name, '性别': s.gender,
                        '平时1': s.scores.regular1, '平时2': s.scores.regular2,
                        '专项': ((parseFloat(s.scores.special1) + parseFloat(s.scores.special2))/2).toFixed(1),
                        '统考': ((parseFloat(s.scores.stdItem1) + parseFloat(s.scores.stdItem2))/2).toFixed(1),
                        '乐跑(次数)': s.scores.runCount, '乐跑(成绩)': s.scores.runScore,
                    };
                    if (semType === 'spring') row['理论'] = s.scores.theory;
                    row['总评'] = finalScore;
                    return row;
                });
                const ws = XLSX.utils.json_to_sheet(exportData);
                const wb = XLSX.utils.book_new();
                XLSX.utils.book_append_sheet(wb, ws, "成绩表");
                XLSX.writeFile(wb, `${semester}-${selectedClass}-${currentUser.name}.xlsx`);
            };

            const batchOptions = [
                { key: 'regular1', label: '平时分1' },
                { key: 'regular2', label: '平时分2' },
                { key: 'special1', label: '专项分1' },
                { key: 'special2', label: '专项分2' },
                { key: 'runCount', label: '乐跑次数' },
                { key: 'runScore', label: '乐跑得分' },
                { key: 'stdItem1', label: '统考项目1' }, // 这里的 Key 用于识别
                { key: 'stdItem2', label: '统考项目2' },
            ];
            if (semType === 'spring') batchOptions.push({ key: 'theory', label: '理论成绩' });

            return (
                <div className="min-h-screen pb-20">
                    <header className="bg-white shadow sticky top-0 z-30">
                        <div className="max-w-7xl mx-auto px-4 py-3 flex justify-between items-center">
                            <div>
                                <h1 className="text-lg font-bold text-gray-800">{currentUser.name}老师</h1>
                                <div className="mt-1">
                                    <select value={semester} onChange={(e) => setSemester(e.target.value)} className="text-xs text-gray-500 border rounded p-1 bg-gray-50">
                                        {semesterOptions.map(opt => (<option key={opt} value={opt}>{opt} ({opt.endsWith('1') ? '秋季' : '春季'})</option>))}
                                    </select>
                                </div>
                            </div>
                            <div className="flex items-center gap-2">
                                <button onClick={onLogout} className="text-sm text-gray-500 hover:text-red-500">退出</button>
                            </div>
                        </div>
                        <div className="px-4 py-2 bg-gray-50 border-t overflow-x-auto whitespace-nowrap hide-scrollbar flex items-center">
                            <button onClick={() => setShowClaimModal(true)} className="mr-3 px-3 py-1.5 rounded-full text-sm border border-dashed border-blue-400 text-blue-600 bg-blue-50 flex items-center gap-1 hover:bg-blue-100">
                                认领班级
                            </button>
                            {myClasses.map(cls => (
                                <button key={cls} onClick={() => setSelectedClass(cls)} className={`mr-3 px-4 py-1.5 rounded-full text-sm transition-all ${selectedClass === cls ? 'bg-blue-600 text-white shadow' : 'bg-white border text-gray-600'}`}>
                                    {cls}
                                </button>
                            ))}
                        </div>
                    </header>

                    <main className="max-w-7xl mx-auto px-4 py-4">
                        {!selectedClass ? (
                            <div className="text-center mt-20 text-gray-400">
                                <Icon name="clipboard-list" className="mx-auto mb-2 h-12 w-12 opacity-50"/>
                                <p>请选择班级或点击“认领班级”添加新班级</p>
                            </div>
                        ) : (
                            <div>
                                <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center mb-4 gap-4">
                                    <span className="text-sm text-gray-500">班级人数: {classStudents.length}</span>
                                    
                                    <div className="flex gap-2">
                                        <div className="flex bg-white rounded-lg shadow-sm border p-1">
                                            <button onClick={() => setViewMode('card')} className={`px-3 py-1 text-xs rounded ${viewMode === 'card' ? 'bg-blue-100 text-blue-700 font-bold' : 'text-gray-500'}`}>卡片模式</button>
                                            <button onClick={() => setViewMode('batch')} className={`px-3 py-1 text-xs rounded ${viewMode === 'batch' ? 'bg-blue-100 text-blue-700 font-bold' : 'text-gray-500'}`}>按项目录入</button>
                                        </div>
                                        
                                        <button onClick={handleExport} className="flex items-center gap-1 bg-green-600 text-white px-3 py-1.5 rounded text-sm hover:bg-green-700">
                                            <Icon name="download" size={14}/> 导出
                                        </button>
                                    </div>
                                </div>

                                {viewMode === 'batch' && (
                                    <div className="mb-4 flex items-center gap-2">
                                        <label className="text-sm font-bold text-gray-700">录入项目:</label>
                                        <select 
                                            value={batchField} 
                                            onChange={(e) => setBatchField(e.target.value)}
                                            className="border border-blue-300 rounded px-2 py-1 text-sm focus:ring-2 ring-blue-500 outline-none"
                                        >
                                            {batchOptions.map(opt => <option key={opt.key} value={opt.key}>{opt.label}</option>)}
                                        </select>
                                    </div>
                                )}

                                {viewMode === 'card' ? (
                                    <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">
                                        {classStudents.map(s => {
                                            const final = calculateFinalScore(s.scores, semType);
                                            return (
                                                <div key={s.id} onClick={() => setEditingStudent(s)} className="bg-white p-4 rounded-xl shadow-sm border border-gray-100 hover:shadow-md transition-shadow cursor-pointer active:scale-95 duration-100">
                                                    <div className="flex justify-between items-start mb-2">
                                                        <div><h3 className="font-bold text-gray-800">{s.name}</h3><p className="text-xs text-gray-500">{s.id} | {s.gender}</p></div>
                                                        <div className={`text-lg font-bold ${parseFloat(final) < 60 ? 'text-red-500' : 'text-green-600'}`}>{final}</div>
                                                    </div>
                                                    <div className="flex gap-2 text-xs text-gray-400 mt-2 border-t pt-2">
                                                        <span>乐跑:{s.scores.runScore}</span><span>专项:{(parseFloat(s.scores.special1)+parseFloat(s.scores.special2))/2}</span>
                                                    </div>
                                                </div>
                                            );
                                        })}
                                    </div>
                                ) : (
                                    <BatchEntryTable 
                                        students={classStudents} 
                                        field={batchField} 
                                        fieldName={batchOptions.find(o => o.key === batchField)?.label} 
                                        semester={semester}
                                        onBatchSave={handleBatchSaveScore}
                                    />
                                )}
                            </div>
                        )}
                    </main>

                    {showClaimModal && (
                        <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-[100] p-4">
                            <div className="bg-white rounded-xl w-full max-w-lg shadow-2xl flex flex-col max-h-[80vh] relative z-[101]">
                                <div className="p-4 border-b flex justify-between items-center bg-gray-50 rounded-t-xl">
                                    <h3 className="text-lg font-bold">认领/添加班级</h3>
                                    <button onClick={() => setShowClaimModal(false)} className="text-2xl">&times;</button>
                                </div>
                                <div className="p-4 overflow-y-auto">
                                    <p className="text-sm text-gray-500 mb-4">列表已按名称排序。点击“添加”即可将该班级学生分配到您名下。</p>
                                    <div className="space-y-2">
                                        {allClassesForClaim.map(cls => (
                                            <div key={cls.name} className="flex justify-between items-center p-3 border rounded hover:bg-gray-50">
                                                <div>
                                                    <div className="font-bold text-gray-800">{cls.name}</div>
                                                    <div className="text-xs text-gray-500">
                                                        学生数: {cls.count} | 
                                                        当前教师: {cls.teacher ? <span className="text-green-600">{cls.teacher}</span> : <span className="text-red-500">未分配</span>}
                                                    </div>
                                                </div>
                                                <button 
                                                    type="button"
                                                    onClick={(e) => { e.stopPropagation(); handleClaimClass(cls.name); }}
                                                    className={`text-xs px-3 py-1.5 rounded text-white ${cls.teacher === currentUser.name ? 'bg-gray-400 cursor-not-allowed' : 'bg-blue-600 hover:bg-blue-700'}`}
                                                    disabled={cls.teacher === currentUser.name}
                                                >
                                                    {cls.teacher === currentUser.name ? '已认领' : '添加'}
                                                </button>
                                            </div>
                                        ))}
                                    </div>
                                </div>
                            </div>
                        </div>
                    )}

                    {editingStudent && (
                        <GradingModal 
                            student={editingStudent} 
                            semester={semester} 
                            onClose={() => setEditingStudent(null)} 
                            onSave={handleSaveFullScore}
                        />
                    )}
                    
                    {toastMsg && <Toast message={toastMsg} onClose={() => setToastMsg('')} />}
                </div>
            );
        };

        const App = () => {
            const [user, setUser] = useState(null);
            const [students, setStudents] = useState(() => {
                const saved = localStorage.getItem('pe_students');
                return saved ? JSON.parse(saved) : INITIAL_STUDENTS;
            });
            const [teachers, setTeachers] = useState(() => {
                const saved = localStorage.getItem('pe_teachers');
                return saved ? JSON.parse(saved) : INITIAL_TEACHERS;
            });
            const [semester, setSemester] = useState(() => {
                return localStorage.getItem('pe_semester') || getDefaultSemester();
            });

            useEffect(() => { localStorage.setItem('pe_students', JSON.stringify(students)); }, [students]);
            useEffect(() => { localStorage.setItem('pe_teachers', JSON.stringify(teachers)); }, [teachers]);
            useEffect(() => { localStorage.setItem('pe_semester', semester); }, [semester]);
            useEffect(() => { if(window.lucide) window.lucide.createIcons(); }, []);

            if (!user) return <LoginPage onLogin={setUser} users={teachers} />;

            return user.role === 'admin' 
                ? <AdminPanel students={students} setStudents={setStudents} teachers={teachers} setTeachers={setTeachers} semester={semester} setSemester={setSemester} onLogout={() => setUser(null)} />
                : <TeacherPanel currentUser={user} students={students} setStudents={setStudents} semester={semester} setSemester={setSemester} onLogout={() => setUser(null)} />;
        };

        const root = ReactDOM.createRoot(document.getElementById('root'));
        root.render(<App />);
    </script>
